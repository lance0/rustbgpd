  >diagnostic.jsonl 2>stderr.log

# Each diagnostic must retain the exact operation and fixture contract.
jq -e -s '
  length == 2 and
  all(.[]; .schema_version == 1 and .operations == 10000) and
  (map(select(.benchmark == "attr_encode/rich/11")) | length == 1) and
  (map(select(.benchmark == "validate_update")) | length == 1)
' diagnostic.jsonl

# Harness red proof: this mutation must make the preceding jq command fail.
sed 's/"operations":10000/"operations":9999/' diagnostic.jsonl \
  >diagnostic-9999.jsonl
